BEGIN_C
#include <stdio.h>
#include <string.h>

int main(void)
{
    char buf[512];
    size_t len;
    size_t i;

    if (fgets(buf, sizeof buf, stdin) == NULL) {
        buf[0] = '\0';
    }

    len = strlen(buf);
    if (len > 0 && buf[len - 1] == '\n') {
        buf[len - 1] = '\0';
        len--;
    }

    for (i = 0; i < len / 2; i++) {
        char t = buf[i];
        buf[i] = buf[len - 1 - i];
        buf[len - 1 - i] = t;
    }

    fwrite(buf, 1, len, stdout);
    putchar('\n');
    fflush(stdout);
    return 0;
}
END_C
