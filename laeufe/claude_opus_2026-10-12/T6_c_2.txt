BEGIN_C
#include <stdio.h>
#include <stdlib.h>

static int cmp(const void *a, const void *b)
{
    int x = *(const int *)a;
    int y = *(const int *)b;
    return (x > y) - (x < y);
}

int main(void)
{
    int a[100];
    int n = 0;
    int v;
    while (n < 100 && scanf("%d", &v) == 1) {
        a[n++] = v;
    }
    qsort(a, n, sizeof(int), cmp);
    for (int i = 0; i < n; i++) {
        printf("%d\n", a[i]);
    }
    return 0;
}
END_C
