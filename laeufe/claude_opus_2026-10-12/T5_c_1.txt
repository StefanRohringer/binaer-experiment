BEGIN_C
#include <stdio.h>
#include <stdlib.h>

int main(int argc, char **argv)
{
    if (argc < 2) {
        fprintf(stderr, "usage: %s FILE\n", argv[0]);
        return 1;
    }

    FILE *f = fopen(argv[1], "rb");
    if (f == NULL) {
        perror(argv[1]);
        return 1;
    }

    unsigned char buf[65536];
    unsigned long count = 0;
    size_t n;

    while ((n = fread(buf, 1, sizeof buf, f)) > 0) {
        for (size_t i = 0; i < n; i++) {
            if (buf[i] == '\n')
                count++;
        }
    }

    if (ferror(f)) {
        perror(argv[1]);
        fclose(f);
        return 1;
    }

    fclose(f);
    printf("%lu\n", count);
    return 0;
}
END_C
