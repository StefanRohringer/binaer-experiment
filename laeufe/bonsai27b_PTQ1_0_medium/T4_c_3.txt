BEGIN_C
#include <stdio.h>

int main(void) {
    char buf[201];
    if (fgets(buf, sizeof(buf), stdin) == NULL) {
        return 0;
    }

    int len = 0;
    while (buf[len] != '\0') {
        len++;
    }

    if (len > 0 && buf[len - 1] == '\n') {
        len--;
    }

    for (int i = len - 1; i >= 0; i--) {
        putchar(buf[i]);
    }
    putchar('\n');

    return 0;
}
END_C
