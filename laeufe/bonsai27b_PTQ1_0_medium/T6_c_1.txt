BEGIN_C
#include <stdio.h>

int main(void)
{
    int nums[100];
    int count = 0;
    int x;

    while (count < 100 && scanf("%d", &x) == 1) {
        nums[count++] = x;
    }

    for (int i = 0; i < count - 1; i++) {
        for (int j = 0; j < count - 1 - i; j++) {
            if (nums[j] > nums[j + 1]) {
                int tmp = nums[j];
                nums[j] = nums[j + 1];
                nums[j + 1] = tmp;
            }
        }
    }

    for (int i = 0; i < count; i++) {
        printf("%d\n", nums[i]);
    }

    return 0;
}
END_C
