#include <stdio.h>
#include <stdlib.h>
#include <sys/stat.h>

int main(int argc, char *argv[]) {

    }

    if (fp == NULL) {
        perror(argv[1]);
        return 1;
    }
    
    /* Check file size - max 1 MB */
    struct stat st;
    if (fstat(fileno(fp), &st) != 0) {
        perror("fstat");
        fclose(fp);
        return 1;
    }
    
    if (st.st_size > 1024 * 1024) {
        fprintf(stderr, "File too large\n");
        fclose(fp);
        return 1;
    }
    
    long count = 0;
    int ch;
    while ((ch = fgetc(fp)) != EOF) {
        if (ch == '\n') {
            count++;
        }
    }
    
    printf("%ld\n", count);
    fclose(fp);
    return 0;
}
