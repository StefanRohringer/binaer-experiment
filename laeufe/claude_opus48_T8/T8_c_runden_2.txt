Hier ist mein Lösungsansatz: Ein einfacher rekursiver Abstiegsparser mit einer Symboltabelle für Variablen.

BEGIN_C
#include <stdio.h>
#include <stdlib.h>
#include <string.h>
#include <ctype.h>

typedef long long ll;

static char names[30][9];
static ll values[30];
static int nvars = 0;

static int getvar(const char *name) {
    for (int i = 0; i < nvars; i++)
        if (strcmp(names[i], name) == 0)
            return i;
    return -1;
}

static int setvar(const char *name, ll val) {
    int idx = getvar(name);
    if (idx < 0) {
        idx = nvars++;
        strncpy(names[idx], name, 8);
        names[idx][8] = 0;
    }
    values[idx] = val;
    return idx;
}

static const char *p; /* current parse pointer */

static void skipws(void) {
    while (*p == ' ' || *p == '\t')
        p++;
}

static ll parseExpr(void);

/* parse a primary: number, name, or (expr) */
static ll parsePrimary(void) {
    skipws();
    if (*p == '(') {
        p++;
        ll v = parseExpr();
        skipws();
        if (*p == ')')
            p++;
        return v;
    }
    if (isdigit((unsigned char)*p)) {
        ll v = 0;
        while (isdigit((unsigned char)*p)) {
            v = v * 10 + (*p - '0');
            p++;
        }
        return v;
    }
    if (*p >= 'a' && *p <= 'z') {
        char name[9];
        int n = 0;
        while (*p >= 'a' && *p <= 'z' && n < 8) {
            name[n++] = *p;
            p++;
        }
        name[n] = 0;
        int idx = getvar(name);
        if (idx < 0)
            return 0;
        return values[idx];
    }
    return 0;
}

/* parse * / % */
static ll parseTerm(void) {
    ll v = parsePrimary();
    for (;;) {
        skipws();
        char op = *p;
        if (op == '*' || op == '/' || op == '%') {
            p++;
            ll r = parsePrimary();
            if (op == '*') v = v * r;
            else if (op == '/') v = v / r;
            else v = v % r;
        } else break;
    }
    return v;
}

/* parse + - */
static ll parseExpr(void) {
    ll v = parseTerm();
    for (;;) {
        skipws();
        char op = *p;
        if (op == '+' || op == '-') {
            p++;
            ll r = parseTerm();
            if (op == '+') v = v + r;
            else v = v - r;
        } else break;
    }
    return v;
}

int main(void) {
    char line[256];
    while (fgets(line, sizeof(line), stdin)) {
        /* strip newline */
        size_t len = strlen(line);
        while (len > 0 && (line[len-1] == '\n' || line[len-1] == '\r'))
            line[--len] = 0;

        p = line;
        skipws();
        if (*p == 0)
            continue; /* empty line */

        /* check for "print" */
        if (strncmp(p, "print", 5) == 0 &&
            !((p[5] >= 'a' && p[5] <= 'z'))) {
            p += 5;
            ll v = parseExpr();
            printf("%lld\n", v);
            continue;
        }

        /* otherwise assignment: name = expr */
        char name[9];
        int n = 0;
        while (*p >= 'a' && *p <= 'z' && n < 8) {
            name[n++] = *p;
            p++;
        }
        name[n] = 0;
        skipws();
        if (*p == '=') {
            p++;
            ll v = parseExpr();
            setvar(name, v);
        }
    }
    return 0;
}
END_C
